cmake_minimum_required(VERSION 3.22.1)
project(zhuoran_chewing LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
get_filename_component(REPO "${CMAKE_CURRENT_SOURCE_DIR}/../../../.." ABSOLUTE)
if(ANDROID_ABI STREQUAL "arm64-v8a")
 set(RUST_TARGET aarch64-linux-android)
elseif(ANDROID_ABI STREQUAL "x86_64")
 set(RUST_TARGET x86_64-linux-android)
else()
 message(FATAL_ERROR "Unsupported ABI")
endif()
add_library(chewing_core STATIC IMPORTED)
set_target_properties(chewing_core PROPERTIES IMPORTED_LOCATION "${REPO}/vendor/libchewing/target/${RUST_TARGET}/release/libchewing_capi.a")
add_library(zhuoran_chewing SHARED chewing_jni.cpp)
target_include_directories(zhuoran_chewing PRIVATE "${REPO}/vendor/libchewing/capi/include")
target_compile_options(zhuoran_chewing PRIVATE -Wall -Wextra -Werror)
target_link_options(zhuoran_chewing PRIVATE "-Wl,-z,max-page-size=16384")
target_link_libraries(zhuoran_chewing PRIVATE chewing_core log dl m)
